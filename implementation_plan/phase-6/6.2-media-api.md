## Step 2: Azure Blob Storage Integration

```csharp
var builder = WebApplication.CreateBuilder(args);

builder.AddServiceDefaults();

// Azure Blob Storage (Aspire — uses Azurite locally, Azure Storage in cloud)
builder.AddAzureBlobServiceClient("blobs");

// MassTransit
builder.Services.AddMassTransit(x =>
{
    x.UsingRabbitMq((context, cfg) =>
    {
        cfg.Host(builder.Configuration.GetConnectionString("messaging"));
        cfg.ConfigureEndpoints(context);
    });

// Image processing service
builder.Services.AddScoped<ImageProcessingService>();

builder.Services.AddEndpointsApiExplorer();
builder.Services.AddSwaggerGen();

// JWT Bearer auth (same pattern as other services)
builder.Services.AddAuthentication("Bearer")
    .AddJwtBearer("Bearer", options => { /* same JWT config */ });
builder.Services.AddAuthorization();

builder.Services.AddOpenApi();

var app = builder.Build();

app.UseMiddleware<GlobalExceptionMiddleware>();
app.MapDefaultEndpoints();
app.UseAuthentication();
app.UseAuthorization();
app.MapOpenApi();

app.MapMediaEndpoints();

app.Run();
```

## Step 3: Media Endpoints

### `MediaEndpoints.cs`

```
POST   /api/media/upload              → Upload file (returns media URL)
GET    /api/media/{blobName}           → Retrieve file (stream response)
GET    /api/media/{blobName}/thumbnail → Retrieve thumbnail
DELETE /api/media/{blobName}           → Delete file
```

### Upload endpoint implementation:
```csharp
group.MapPost("/upload", async (
    IFormFile file,
    BlobServiceClient blobClient,
    ImageProcessingService imageService,
    CancellationToken ct) =>
{
    // Validate file type (jpg, png, gif, webp)
    // Validate file size (max 10MB)
    var containerClient = blobClient.GetBlobContainerClient("media");
    await containerClient.CreateIfNotExistsAsync(cancellationToken: ct);

    var blobName = $"{Guid.NewGuid()}{Path.GetExtension(file.FileName)}";
    var blobClient2 = containerClient.GetBlobClient(blobName);

    using var stream = file.OpenReadStream();
    await blobClient2.UploadAsync(stream, overwrite: true, cancellationToken: ct);

    // If image, generate and upload thumbnail
    if (IsImageFile(file.ContentType))
    {
        var thumbnailName = $"thumb_{blobName}";
        var thumbStream = await imageService.CreateThumbnailAsync(stream, ct);
        var thumbBlob = containerClient.GetBlobClient(thumbnailName);
        await thumbBlob.UploadAsync(thumbStream, overwrite: true, cancellationToken: ct);
    }

    return Results.Created($"/api/media/{blobName}", new { blobName, url = blobClient2.Uri });
})
.DisableAntiforgery();  // File upload from forms
```

### Retrieve endpoint:
```csharp
group.MapGet("/{blobName}", async (
    string blobName,
    BlobServiceClient blobClient,
    CancellationToken ct) =>
{
    var containerClient = blobClient.GetBlobContainerClient("media");
    var blob = containerClient.GetBlobClient(blobName);

    if (!await blob.ExistsAsync(ct))
        return Results.NotFound();

    var download = await blob.DownloadStreamingAsync(cancellationToken: ct);
    return Results.File(download.Value.Content, download.Value.Details.ContentType);
});
```

### Delete endpoint:
```csharp
group.MapDelete("/{blobName}", async (
    string blobName,
    BlobServiceClient blobClient,
    CancellationToken ct) =>
{
    var containerClient = blobClient.GetBlobContainerClient("media");
    var blob = containerClient.GetBlobClient(blobName);
    await blob.DeleteIfExistsAsync(cancellationToken: ct);
    return Results.NoContent();
}).RequireAuthorization();
```

## Step 4: Image Processing Service

### `ImageProcessingService.cs`
```csharp
public sealed class ImageProcessingService
{
    private const int ThumbnailMaxWidth = 200;
    private const int ThumbnailMaxHeight = 200;

    public async Task<Stream> CreateThumbnailAsync(
        Stream inputStream, CancellationToken ct)
    {
        var output = new MemoryStream();
        using var image = await SixLabors.ImageSharp.Image.LoadAsync(inputStream, ct);

        image.Mutate(x => x.Resize(new ResizeOptions
        {
            Size = new Size(ThumbnailMaxWidth, ThumbnailMaxHeight),
            Mode = ResizeMode.Max
        }));

        await image.SaveAsJpegAsync(output, ct);
        output.Position = 0;
        return output;
    }
}
```
