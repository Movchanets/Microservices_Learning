# 7.1.2 — Identity Frontend (SignalStore & BFF)

> **Ref**: [plans/11-frontend-angular.md](../../plans/11-frontend-angular.md) · [plans/06-api-gateway-bff.md](../../plans/06-api-gateway-bff.md)

## Goal
Implement the feature-scoped `AuthStore` using NgRx SignalStore to manage authentication state, and integrate with the YARP BFF using `withCredentials: true`.

## Prerequisites
- Sub-plan 7.1.1 (Identity Auth UI) completed
- NgRx SignalStore installed (`npm i @ngrx/signals`)

## Steps

### Step 1: Create Auth Service
File: `src/web/src/app/core/auth/auth.service.ts`
```typescript
import { Injectable, inject } from '@angular/core';
import { HttpClient } from '@angular/common/http';
import { firstValueFrom } from 'rxjs';
import { LoginCredentials, RegisterCredentials, User } from './auth.models';

@Injectable({ providedIn: 'root' })
export class AuthService {
  private http = inject(HttpClient);
  private baseUrl = '/bff';

  // Note: Interceptor must attach withCredentials: true to all requests
  
  login(credentials: LoginCredentials): Promise<void> {
    // BFF handles the cookie set
    return firstValueFrom(this.http.post<void>(`/api/identity/auth/login`, credentials));
  }

  register(credentials: RegisterCredentials): Promise<void> {
    return firstValueFrom(this.http.post<void>(`/api/identity/auth/register`, credentials));
  }

  logout(): Promise<void> {
    return firstValueFrom(this.http.get<void>(`${this.baseUrl}/logout`));
  }

  getUser(): Promise<User> {
    return firstValueFrom(this.http.get<User>(`${this.baseUrl}/user`));
  }
}
```

### Step 2: Create SignalStore
File: `src/web/src/app/core/auth/auth.store.ts`
```typescript
import { inject } from '@angular/core';
import { patchState, signalStore, withMethods, withState } from '@ngrx/signals';
import { AuthService } from './auth.service';
import { User, LoginCredentials } from './auth.models';
import { Router } from '@angular/router';

type AuthState = {
  user: User | null;
  loading: boolean;
  error: string | null;
};

const initialState: AuthState = {
  user: null,
  loading: false,
  error: null,
};

export const AuthStore = signalStore(
  { providedIn: 'root' }, // Global store for auth
  withState(initialState),
  withMethods((store, authService = inject(AuthService), router = inject(Router)) => ({
    async login(credentials: LoginCredentials) {
      patchState(store, { loading: true, error: null });
      try {
        await authService.login(credentials);
        const user = await authService.getUser();
        patchState(store, { user, loading: false });
        router.navigate(['/']);
      } catch (err: any) {
        patchState(store, { error: err.message, loading: false });
      }
    },
    async checkAuth() {
      patchState(store, { loading: true });
      try {
        const user = await authService.getUser();
        patchState(store, { user, loading: false });
      } catch {
        patchState(store, { user: null, loading: false });
      }
    }
  }))
);
```

### Step 3: Configure HTTP Interceptor
File: `src/web/src/app/core/http/api.interceptor.ts`
```typescript
import { HttpInterceptorFn } from '@angular/common/http';

export const apiInterceptor: HttpInterceptorFn = (req, next) => {
  // Always include credentials for BFF cookie flow
  const apiReq = req.clone({
    withCredentials: true
  });
  return next(apiReq);
};
```

Register in `app.config.ts`:
```typescript
provideHttpClient(withInterceptors([apiInterceptor]))
```
