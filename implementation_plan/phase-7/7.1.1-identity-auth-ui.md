# 7.1.1 — Identity Frontend (Auth UI)

> **Ref**: [plans/11-frontend-angular.md](../../plans/11-frontend-angular.md) · [src/web/Design.md](../../src/web/Design.md)

## Goal
Implement the Login and Registration UI components using Angular 19+ standalone components, Signal Forms, and Spartan UI primitives, adhering to the "Enterprise Gateway" / "Liquid Glass" design system.

## Prerequisites
- Angular application scaffolded
- Tailwind CSS and Spartan UI installed

## Steps

### Step 1: Scaffold Auth Components
Use Angular CLI to generate standalone components.
```bash
ng g c features/auth/login --standalone
ng g c features/auth/register --standalone
```

### Step 2: Implement Login Component (Signal Forms)
File: `src/web/src/app/features/auth/login/login.component.ts`
```typescript
import { Component, ChangeDetectionStrategy, inject } from '@angular/core';
import { FormBuilder, ReactiveFormsModule, Validators } from '@angular/forms';
import { RouterLink } from '@angular/router';

@Component({
  selector: 'app-login',
  standalone: true,
  changeDetection: ChangeDetectionStrategy.OnPush,
  imports: [ReactiveFormsModule, RouterLink],
  template: `
    <div class="min-h-screen bg-background flex items-center justify-center p-4">
      <div class="w-full max-w-md bg-white/70 dark:bg-slate-900/70 backdrop-blur-md rounded-2xl shadow-xl p-8 border border-border">
        <h1 class="font-lexend text-2xl font-semibold text-foreground mb-6 text-center">Welcome Back</h1>
        <form [formGroup]="loginForm" (ngSubmit)="onSubmit()" class="space-y-4">
          <div>
            <label class="block text-sm font-medium text-muted mb-1" i18n>Email</label>
            <input type="email" formControlName="email" class="w-full px-4 py-2 rounded-xl border border-border focus:ring-2 focus:ring-primary focus:border-transparent outline-none transition-all" />
          </div>
          <div>
            <label class="block text-sm font-medium text-muted mb-1" i18n>Password</label>
            <input type="password" formControlName="password" class="w-full px-4 py-2 rounded-xl border border-border focus:ring-2 focus:ring-primary focus:border-transparent outline-none transition-all" />
          </div>
          <button type="submit" [disabled]="loginForm.invalid" class="w-full bg-primary hover:bg-secondary text-white font-medium py-2.5 rounded-xl transition-all scale-100 hover:scale-[1.02] cursor-pointer">
            <span i18n>Sign In</span>
          </button>
        </form>
        <p class="mt-4 text-center text-sm text-muted">
          <span i18n>Don't have an account?</span> 
          <a routerLink="/register" class="text-primary hover:underline" i18n>Register here</a>
        </p>
      </div>
    </div>
  `
})
export class LoginComponent {
  private fb = inject(FormBuilder);
  
  // Use Signal Forms approach if available, or fallback to standard reactive forms
  loginForm = this.fb.group({
    email: ['', [Validators.required, Validators.email]],
    password: ['', Validators.required]
  });

  onSubmit() {
    if (this.loginForm.valid) {
      // Logic handled by Identity Store
    }
  }
}
```

### Step 3: Implement Register Component
Follow the same structure for `register.component.ts`, adding fields for FirstName, LastName, and a password strength validator. Use the same Liquid Glass aesthetic (`backdrop-blur-md`, `bg-white/70`, `rounded-2xl`).

### Step 4: Route Configuration
File: `src/web/src/app/app.routes.ts`
```typescript
import { Routes } from '@angular/router';

export const routes: Routes = [
  {
    path: 'login',
    loadComponent: () => import('./features/auth/login/login.component').then(m => m.LoginComponent)
  },
  {
    path: 'register',
    loadComponent: () => import('./features/auth/register/register.component').then(m => m.RegisterComponent)
  }
];
```
