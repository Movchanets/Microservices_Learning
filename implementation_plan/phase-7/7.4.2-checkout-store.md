# 7.4.2 — Checkout & Order SignalStore

> **Ref**: [plans/11-frontend-angular.md](../../plans/11-frontend-angular.md) · [angular-best-practices]

## Goal
Implement the NgRx SignalStore for the Checkout flow and Order history. The checkout store manages the submission process and tracks the correlation ID. The order store manages order history and detail retrieval.

## Prerequisites
- Sub-plan 7.4.1 completed (models and services)

## Steps

### Step 1: Create Checkout SignalStore

File: `src/web/src/app/features/checkout/checkout.store.ts`
```typescript
import { computed, inject } from '@angular/core';
import { patchState, signalStore, withComputed, withMethods, withState } from '@ngrx/signals';
import { CheckoutService } from './checkout.service';
import { Order } from './checkout.models';
import { CartStore } from '../cart/cart.store';

interface CheckoutState {
  submitting: boolean;
  error: string | null;
  correlationId: string | null;
  order: Order | null;
}

const initialState: CheckoutState = {
  submitting: false,
  error: null,
  correlationId: null,
  order: null,
};

export const CheckoutStore = signalStore(
  { providedIn: 'root' },
  withState(initialState),

  withComputed((store) => ({
    isSubmitted: computed(() => store.correlationId() !== null),
    hasOrder: computed(() => store.order() !== null),
    orderStatus: computed(() => store.order()?.status ?? null),
  })),

  withMethods((
    store,
    checkoutService = inject(CheckoutService),
    cartStore = inject(CartStore),
  ) => ({

    async submitCheckout(): Promise<void> {
      const buyerId = localStorage.getItem('buyerId') || 'guest-user';
      const items = cartStore.items();

      if (items.length === 0) {
        patchState(store, { error: 'Cart is empty' });
        return;
      }

      patchState(store, { submitting: true, error: null });

      try {
        const response = await checkoutService.checkout(buyerId, items);
        patchState(store, {
          correlationId: response.correlationId,
          submitting: false,
        });
        // Clear the cart after successful checkout
        cartStore.clearCartAfterCheckout();
      } catch (err: any) {
        patchState(store, {
          error: 'Checkout failed. Please try again.',
          submitting: false,
        });
      }
    },

    reset(): void {
      patchState(store, initialState);
    },
  }))
);
```

### Step 2: Create Order SignalStore

File: `src/web/src/app/features/orders/order.store.ts`
```typescript
import { computed, inject } from '@angular/core';
import { patchState, signalStore, withComputed, withMethods, withState, withHooks } from '@ngrx/signals';
import { OrderService } from './order.service';
import { Order, OrderStatus } from '../checkout/checkout.models';

interface OrderState {
  orders: Order[];
  selectedOrder: Order | null;
  loading: boolean;
  error: string | null;
}

const initialState: OrderState = {
  orders: [],
  selectedOrder: null,
  loading: false,
  error: null,
};

export const OrderStore = signalStore(
  { providedIn: 'root' },
  withState(initialState),

  withComputed((store) => ({
    completedOrders: computed(() =>
      store.orders().filter(o => o.status === 'Completed')
    ),
    activeOrders: computed(() =>
      store.orders().filter(o =>
        o.status === 'Submitted' ||
        o.status === 'InventoryReserved' ||
        o.status === 'PaymentProcessing'
      )
    ),
    hasOrders: computed(() => store.orders().length > 0),
  })),

  withMethods((store, orderService = inject(OrderService)) => ({

    async loadOrders(buyerId: string): Promise<void> {
      patchState(store, { loading: true, error: null });
      try {
        const orders = await orderService.getOrdersByBuyer(buyerId);
        patchState(store, { orders, loading: false });
      } catch (err: any) {
        patchState(store, { error: 'Failed to load orders', loading: false });
      }
    },

    async loadOrderById(orderId: string): Promise<void> {
      patchState(store, { loading: true, error: null });
      try {
        const order = await orderService.getOrderById(orderId);
        patchState(store, { selectedOrder: order, loading: false });
      } catch (err: any) {
        patchState(store, { error: 'Failed to load order details', loading: false });
      }
    },

    updateOrderStatus(orderId: string, status: OrderStatus): void {
      patchState(store, {
        orders: store.orders().map(o =>
          o.id === orderId ? { ...o, status } : o
        ),
        selectedOrder: store.selectedOrder()?.id === orderId
          ? { ...store.selectedOrder()!, status }
          : store.selectedOrder(),
      });
    },

    clearSelected(): void {
      patchState(store, { selectedOrder: null });
    },
  }))
);
```

### Step 3: Add `clearCartAfterCheckout` method to CartStore

Update: `src/web/src/app/features/cart/cart.store.ts`

Add this method inside `withMethods`:

clearCartAfterCheckout(): void {
  patchState(store, { items: [], checkoutCorrelationId: store.checkoutCorrelationId() });
},
