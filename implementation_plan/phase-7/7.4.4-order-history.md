### Step 2: Create Status Badge Component

File: `src/web/src/app/features/orders/components/status-badge/status-badge.ts`
```typescript
import { Component, ChangeDetectionStrategy, input } from '@angular/core';
import { OrderStatus } from '../../../checkout/checkout.models';

@Component({
  selector: 'app-status-badge',
  standalone: true,
  changeDetection: ChangeDetectionStrategy.OnPush,
  template: `
    <span [class]="badgeClass()">
      {{ label() }}
    </span>
  `
})
export class StatusBadgeComponent {
  status = input.required<OrderStatus>();

  label(): string {
    const labels: Record<OrderStatus, string> = {
      Submitted: 'Submitted',
      InventoryReserved: 'Reserved',
      PaymentProcessing: 'Processing',
      Completed: 'Completed',
      Cancelled: 'Cancelled',
      Faulted: 'Failed',
    };
    return labels[this.status()];
  }

  badgeClass(): string {
    const base = 'inline-flex items-center px-3 py-1 rounded-full text-xs font-semibold';
    const variants: Record<OrderStatus, string> = {
      Submitted: `${base} bg-blue-500/10 text-blue-500`,
      InventoryReserved: `${base} bg-yellow-500/10 text-yellow-500`,
      PaymentProcessing: `${base} bg-primary/10 text-primary`,
      Completed: `${base} bg-green-500/10 text-green-500`,
      Cancelled: `${base} bg-muted/20 text-muted`,
      Faulted: `${base} bg-red-500/10 text-red-500`,
    };
    return variants[this.status()];
  }
}
```

### Step 3: Create Order List Component

File: `src/web/src/app/features/orders/order-list/order-list.ts`
```typescript
import { Component, ChangeDetectionStrategy, inject, OnInit } from '@angular/core';
import { RouterLink } from '@angular/router';
import { LucideAngularModule } from 'lucide-angular';
import { OrderStore } from '../order.store';
import { StatusBadgeComponent } from '../components/status-badge/status-badge';

@Component({
  selector: 'app-order-list',
  standalone: true,
  changeDetection: ChangeDetectionStrategy.OnPush,
  imports: [RouterLink, LucideAngularModule, StatusBadgeComponent],
  template: `
    <div class="min-h-screen bg-background p-6 pt-10">
      <div class="container mx-auto max-w-4xl">
        <h1 class="text-3xl font-bold text-foreground font-lexend mb-8" i18n>Your Orders</h1>

        @if (store.loading()) {
          <div class="flex justify-center p-12">
            <div class="animate-spin w-8 h-8 border-4 border-primary border-t-transparent rounded-full"></div>
          </div>
        } @else if (store.error()) {
          <div class="p-4 bg-red-500/10 text-red-500 rounded-xl">
            {{ store.error() }}
          </div>
        } @else if (!store.hasOrders()) {
          <div class="text-center py-16 bg-card/60 backdrop-blur-sm rounded-3xl border border-border">
            <lucide-icon name="Package" class="w-16 h-16 mx-auto mb-4 opacity-30"></lucide-icon>
            <p class="text-xl font-medium text-foreground mb-4" i18n>No orders yet</p>
            <a routerLink="/catalog"
               class="inline-block px-6 py-3 bg-primary text-white rounded-xl hover:bg-secondary transition-colors">
              <span i18n>Start Shopping</span>
            </a>
          </div>
        } @else {
          <!-- Active Orders -->
          @if (store.activeOrders().length > 0) {
            <h2 class="text-lg font-semibold font-lexend mb-4 text-foreground" i18n>Active Orders</h2>
            <div class="space-y-3 mb-8">
              @for (order of store.activeOrders(); track order.id) {
                <a [routerLink]="['/orders', order.id]"
                   class="block bg-card/60 backdrop-blur-sm rounded-2xl border border-border p-5 hover:shadow-lg transition-all cursor-pointer">
                  <div class="flex items-center justify-between">
                    <div>
                      <p class="font-mono text-sm text-muted mb-1">{{ order.id | slice:0:8 }}...</p>
                      <p class="font-medium text-foreground">
                        {{ order.items.length }} {{ order.items.length === 1 ? 'item' : 'items' }}
                      </p>
                    </div>
                    <div class="text-right">
                      <app-status-badge [status]="order.status" />
                      <p class="text-sm text-muted mt-1">{{ order.createdAt | date:'short' }}</p>
                    </div>
                  </div>
                </a>
              }
            </div>
          }

          <!-- Completed / Past Orders -->
          @if (store.completedOrders().length > 0) {
            <h2 class="text-lg font-semibold font-lexend mb-4 text-foreground" i18n>Past Orders</h2>
            <div class="space-y-3">
              @for (order of store.completedOrders(); track order.id) {
                <a [routerLink]="['/orders', order.id]"
                   class="block bg-card/60 backdrop-blur-sm rounded-2xl border border-border p-5 hover:shadow-lg transition-all cursor-pointer">
                  <div class="flex items-center justify-between">
                    <div>
                      <p class="font-mono text-sm text-muted mb-1">{{ order.id | slice:0:8 }}...</p>
                      <p class="font-medium text-foreground">
                        {{ order.items.length }} {{ order.items.length === 1 ? 'item' : 'items' }}
                      </p>
                    </div>
                    <div class="text-right">
                      <app-status-badge [status]="order.status" />
                      <p class="text-sm text-muted mt-1">{{ order.createdAt | date:'short' }}</p>
                    </div>
                  </div>
                </a>
              }
            </div>
          }
        }
      </div>
    </div>
  `
})
export class OrderListComponent implements OnInit {
  readonly store = inject(OrderStore);

  ngOnInit(): void {
    const buyerId = localStorage.getItem('buyerId') || 'guest-user';
    this.store.loadOrders(buyerId);
  }
}
```

### Step 4: Create Order Detail Component

```typescript
import { Component, ChangeDetectionStrategy, inject, OnInit, input } from '@angular/core';
import { RouterLink } from '@angular/router';
import { LucideAngularModule } from 'lucide-angular';
import { OrderStore } from '../order.store';
import { StatusBadgeComponent } from '../components/status-badge/status-badge';

@Component({
  selector: 'app-order-detail',
  standalone: true,
  changeDetection: ChangeDetectionStrategy.OnPush,
  imports: [RouterLink, LucideAngularModule, StatusBadgeComponent],
  template: `
    <div class="min-h-screen bg-background p-6 pt-10">
      <div class="container mx-auto max-w-3xl">
        @if (store.loading()) {
          <div class="flex justify-center p-12">
            <div class="animate-spin w-8 h-8 border-4 border-primary border-t-transparent rounded-full"></div>
          </div>
        } @else if (store.selectedOrder(); as order) {
          <div class="mb-6">
            <a routerLink="/orders" class="text-muted hover:text-foreground transition-colors flex items-center gap-2 mb-4">
              <lucide-icon name="ChevronLeft" class="w-4 h-4"></lucide-icon>
              <span i18n>Back to Orders</span>
            </a>
            <div class="flex items-center justify-between">
              <h1 class="text-2xl font-bold font-lexend" i18n>Order Details</h1>
              <app-status-badge [status]="order.status" />
            </div>
          </div>

          <!-- Order Info Card -->
          <div class="bg-card/60 backdrop-blur-sm rounded-3xl border border-border p-6 mb-6">
            <div class="grid grid-cols-2 gap-4 text-sm">
              <div>
                <p class="text-muted" i18n>Order ID</p>
                <p class="font-mono">{{ order.id }}</p>
              </div>
              <div>
                <p class="text-muted" i18n>Placed On</p>
                <p>{{ order.createdAt | date:'medium' }}</p>
              </div>
              @if (order.completedAt) {
                <div>
                  <p class="text-muted" i18n>Completed On</p>
                  <p>{{ order.completedAt | date:'medium' }}</p>
                </div>
              }
              <div>
                <p class="text-muted" i18n>Total</p>
                <p class="text-xl font-bold font-lexend">{{ order.totalAmount | currency }}</p>
              </div>
            </div>
          </div>

          <!-- Order Items -->
          <div class="bg-card/60 backdrop-blur-sm rounded-3xl border border-border overflow-hidden">
            <h2 class="text-lg font-semibold font-lexend p-6 pb-0" i18n>Items</h2>
            <ul class="divide-y divide-border p-6">
              @for (item of order.items; track item.id) {
                <li class="py-4 flex items-center justify-between">
                  <div class="flex items-center gap-4">
                    <div class="w-12 h-12 bg-muted/20 rounded-xl flex items-center justify-center">
                      <lucide-icon name="Package" class="w-6 h-6 text-muted/50"></lucide-icon>
                    </div>
                    <div>
                      <p class="font-medium">{{ item.productName }}</p>
                      <p class="text-sm text-muted">SKU: {{ item.sku }}</p>
                    </div>
                  </div>
                  <div class="text-right">
                    <p class="font-medium">{{ item.totalPrice | currency }}</p>
                    <p class="text-sm text-muted">Qty: {{ item.quantity }}</p>
                  </div>
                </li>
              }
            </ul>
          </div>
        } @else if (store.error()) {
          <div class="p-4 bg-red-500/10 text-red-500 rounded-xl">
            {{ store.error() }}
          </div>
        }
      </div>
    </div>
  `
})
export class OrderDetailComponent implements OnInit {
  readonly store = inject(OrderStore);
  orderId = input.required<string>();

  ngOnInit(): void {
    this.store.loadOrderById(this.orderId());
  }
}
```

### Step 5: Configure Order Routes

File: `src/web/src/app/features/orders/order.routes.ts`
```typescript
import { Routes } from '@angular/router';

export default [
  {
    path: '',
    loadComponent: () => import('./order-list/order-list').then(c => c.OrderListComponent),
  },
  {
    path: ':id',
    loadComponent: () => import('./order-detail/order-detail').then(c => c.OrderDetailComponent),
  },
] as Routes;
```
