import { Injectable, inject } from '@angular/core';
import { HttpClient } from '@angular/common/http';
import { firstValueFrom } from 'rxjs';
import { Product } from './catalog.models';

@Injectable({ providedIn: 'root' })
export class CatalogService {
  private http = inject(HttpClient);
  
  getProducts(): Promise<Product[]> {
    return firstValueFrom(this.http.get<Product[]>('/api/catalog/products'));
  }
}

### Step 2: Create Catalog SignalStore
File: `src/web/src/app/features/catalog/catalog.store.ts`
```typescript
import { inject, computed } from '@angular/core';
import { patchState, signalStore, withComputed, withMethods, withState } from '@ngrx/signals';
import { rxMethod } from '@ngrx/signals/rxjs-interop';
import { CatalogService } from './catalog.service';
import { Product } from './catalog.models';
import { pipe, switchMap, tap } from 'rxjs';

type CatalogState = {
  products: Product[];
  searchQuery: string;
  loading: boolean;
};

const initialState: CatalogState = {
  products: [],
  searchQuery: '',
  loading: false,
};

export const CatalogStore = signalStore(
  // Scoped to the feature, NOT provided in root
  withState(initialState),
  withComputed((store) => ({
    filteredProducts: computed(() => {
      const query = store.searchQuery().toLowerCase();
      if (!query) return store.products();
      return store.products().filter(p => 
        p.name.toLowerCase().includes(query) || 
        p.description.toLowerCase().includes(query)
      );
    })
  })),
  withMethods((store, catalogService = inject(CatalogService)) => ({
    updateSearchQuery(query: string) {
      patchState(store, { searchQuery: query });
    },
    async loadProducts() {
      patchState(store, { loading: true });
      try {
        const products = await catalogService.getProducts();
        patchState(store, { products, loading: false });
      } catch (err) {
        patchState(store, { loading: false });
        console.error('Failed to load products', err);
      }
    }
  }))
);
```
