# 7.4.3 — Checkout Page UI

> **Ref**: [plans/11-frontend-angular.md](../../plans/11-frontend-angular.md) · [src/web/Design.md](../../src/web/Design.md)

## Goal
Build the Checkout page with order summary, confirmation flow, and real-time status tracking using the `CheckoutStore` and `CartStore`.

## Prerequisites
- Sub-plan 7.4.2 completed (CheckoutStore, OrderStore)
- Sub-plan 7.3.3 completed (Cart page with checkout button)

## Important: Async Order Creation Timing

After `CartStore.checkout()` returns a `correlationId`, the Order entity is created **asynchronously** by the Ordering saga. The order does NOT exist in the Ordering API immediately. The frontend must:

1. Show "Order Submitted" immediately (based on `correlationId` from CartStore)
2. Poll `GET /api/orders/buyer/{buyerId}` to find the newly created order
3. Once found, display the order status from the Order entity

SignalR (sub-plan 7.4.5) will replace polling with real-time push updates.

### Step 4: Create Checkout Page Component

File: `src/web/src/app/features/checkout/checkout-page/checkout-page.ts`
```typescript
import { Component, ChangeDetectionStrategy, inject, OnInit, OnDestroy } from '@angular/core';
import { Router } from '@angular/router';
import { LucideAngularModule } from 'lucide-angular';
import { CartStore } from '../../cart/cart.store';
import { CheckoutStore } from '../checkout.store';
import { OrderStore } from '../../orders/order.store';
import { CheckoutSummaryComponent } from '../checkout-summary/checkout-summary';
import { CheckoutStatusComponent } from '../checkout-status/checkout-status';

@Component({
  selector: 'app-checkout-page',
  standalone: true,
  changeDetection: ChangeDetectionStrategy.OnPush,
  imports: [LucideAngularModule, CheckoutSummaryComponent, CheckoutStatusComponent],
  template: `
    <div class="min-h-screen bg-background p-6 pt-10">
      <div class="container mx-auto max-w-3xl">
        <h1 class="text-3xl font-bold text-foreground font-lexend mb-8" i18n>Checkout</h1>

        @if (checkoutStore.hasOrder()) {
          <!-- Order found — show status tracking -->
          <app-checkout-status
            [status]="checkoutStore.orderStatus()!"
            [correlationId]="cartStore.checkoutCorrelationId()" />
        } @else if (submitted()) {
          <!-- Order submitted but not yet created in Ordering API — polling -->
          <div class="bg-card/60 backdrop-blur-sm rounded-3xl border border-border p-8 text-center">
            <div class="animate-pulse">
              <lucide-icon name="Clock" class="w-16 h-16 mx-auto mb-4 text-primary"></lucide-icon>
              <h2 class="text-2xl font-bold font-lexend mb-2" i18n>Order Submitted</h2>
              <p class="text-muted mb-6" i18n>Your order is being created...</p>
            </div>
            <div class="mt-4">
              <p class="text-sm text-muted mb-2" i18n>Correlation ID</p>
              <p class="font-mono text-xs text-muted bg-muted/10 inline-block px-3 py-1 rounded-lg">
                {{ cartStore.checkoutCorrelationId() }}
              </p>
            </div>
          </div>
        } @else if (cartStore.isEmpty()) {
          <!-- Empty cart -->
          <div class="text-center py-16 bg-card/60 backdrop-blur-sm rounded-3xl border border-border">
            <lucide-icon name="ShoppingCart" class="w-16 h-16 mx-auto mb-4 opacity-30"></lucide-icon>
            <p class="text-xl font-medium text-foreground mb-4" i18n>Your cart is empty</p>
            <a href="/catalog"
               class="inline-block px-6 py-3 bg-primary text-white rounded-xl hover:bg-secondary transition-colors">
              <span i18n>Continue Shopping</span>
            </a>
          </div>
        } @else {
          <!-- Pre-submission: show summary + confirm -->
          <app-checkout-summary />

          @if (checkoutStore.error()) {
            <div class="mt-4 p-4 bg-red-500/10 text-red-500 rounded-xl">
              {{ checkoutStore.error() }}
            </div>
          }

          <div class="mt-6 flex items-center justify-between">
            <a href="/cart" class="text-muted hover:text-foreground transition-colors flex items-center gap-2">
              <lucide-icon name="ChevronLeft" class="w-4 h-4"></lucide-icon>
              <span i18n>Back to Cart</span>
            </a>
            <button (click)="onConfirm()"
                    [disabled]="checkoutStore.submitting()"
                    class="px-8 py-3 bg-success text-white rounded-xl font-medium hover:opacity-90 transition-all disabled:opacity-50 cursor-pointer">
              @if (checkoutStore.submitting()) {
                <span class="flex items-center gap-2">
                  <span class="animate-spin w-4 h-4 border-2 border-white border-t-transparent rounded-full"></span>
                  <span i18n>Processing...</span>
                </span>
              } @else {
                <span i18n>Confirm Order</span>
              }
            </button>
          </div>
        }
      </div>
    </div>
  `
})
export class CheckoutPageComponent implements OnInit, OnDestroy {
  private readonly router = inject(Router);
  readonly cartStore = inject(CartStore);
  readonly checkoutStore = inject(CheckoutStore);
  private readonly orderStore = inject(OrderStore);

  private pollTimer: ReturnType<typeof setInterval> | null = null;
  submitted = false;

  ngOnInit(): void {
    this.checkoutStore.reset();
  }

  ngOnDestroy(): void {
    this.stopPolling();
  }

  async onConfirm(): Promise<void> {
    await this.checkoutStore.submitCheckout();

    if (!this.checkoutStore.error()) {
      this.submitted = true;
      this.startPolling();
    }
  }

  private startPolling(): void {
    const buyerId = localStorage.getItem('buyerId') || 'guest-user';

    // Poll every 2s for the order to appear in the Ordering API
    this.pollTimer = setInterval(async () => {
      await this.orderStore.loadOrders(buyerId);
      const orders = this.orderStore.orders();
      const correlationId = this.cartStore.checkoutCorrelationId();

      // Find the most recent order (the saga creates it with OrderId = CorrelationId)
      const order = orders.find(o => o.id === correlationId) ?? orders[0];

      if (order) {
        this.checkoutStore.setOrder(order);
        this.stopPolling();
      }
    }, 2000);

    // Stop polling after 30s (saga should complete well before)
    setTimeout(() => this.stopPolling(), 30_000);
  }

  private stopPolling(): void {
    if (this.pollTimer) {
      clearInterval(this.pollTimer);
      this.pollTimer = null;
    }
  }
}
```

> **Note**: Polling is a temporary solution. Sub-plan 7.4.5 (SignalR) will replace it with real-time push updates.
