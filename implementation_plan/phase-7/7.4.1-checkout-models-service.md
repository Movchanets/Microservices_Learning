# 7.4.1 — Checkout & Order Models & Services

> **Ref**: [plans/11-frontend-angular.md](../../plans/11-frontend-angular.md) · [plans/06-api-gateway-bff.md](../../plans/06-api-gateway-bff.md) · [src/web/Design.md](../../src/web/Design.md)

## Goal
Define frontend data models for Checkout and Order features. Implement Angular services to communicate with `Cart.API` (checkout) and `Ordering.API` (order queries) endpoints through the BFF gateway.

## Prerequisites
- Phase 4 backend (Ordering.API, Payment.API) is running and accessible via the API Gateway
- Sub-plan 7.3.1 completed (Cart models and service exist)
- Angular HTTP client is configured with `apiInterceptor`

## BFF Routing Rules
- **Checkout**: `POST /api/cart/checkout` → `cart-api` (triggers `OrderSubmittedEvent`)
- **Orders**: `GET /api/orders/{id}`, `GET /api/orders/buyer/{buyerId}` → `ordering-api`
- **Payments**: `GET /api/payments/order/{orderId}` → `payment-api`

## Steps

### Step 1: Create Checkout models

File: `src/web/src/app/features/checkout/checkout.models.ts`
```typescript
import { CartItem } from '../cart/cart.models';

export interface CheckoutRequest {
  buyerId: string;
  items: CartItem[];
}

export interface CheckoutResponse {
  correlationId: string;
}

export type OrderStatus =
  | 'Submitted'
  | 'InventoryReserved'
  | 'PaymentProcessing'
  | 'Completed'
  | 'Cancelled'
  | 'Faulted';

export interface OrderItem {
  id: string;
  sku: string;
  productName: string;
  unitPrice: number;
  quantity: number;
  totalPrice: number;
}

export interface Order {
  id: string;
  buyerId: string;
  status: OrderStatus;
  totalAmount: number;
  createdAt: string;
  completedAt: string | null;
  items: OrderItem[];
}

export interface PaymentStatus {
  id: string;
  orderId: string;
  amount: number;
  status: 'Pending' | 'Completed' | 'Failed' | 'Refunded';
  transactionId: string | null;
  failureReason: string | null;
  createdAt: string;
  processedAt: string | null;
}
```

### Step 2: Create Checkout Service

File: `src/web/src/app/features/checkout/checkout.service.ts`
```typescript
import { Injectable, inject } from '@angular/core';
import { HttpClient } from '@angular/common/http';
import { firstValueFrom } from 'rxjs';
import { CheckoutResponse } from './checkout.models';
import { CartItem } from '../cart/cart.models';

@Injectable({ providedIn: 'root' })
export class CheckoutService {
  private readonly http = inject(HttpClient);

  async checkout(buyerId: string, items: CartItem[]): Promise<CheckoutResponse> {
    return firstValueFrom(
      this.http.post<CheckoutResponse>('/api/cart/checkout', { buyerId, items })
    );
  }
}
```

### Step 3: Create Order Service

File: `src/web/src/app/features/orders/order.service.ts`
```typescript
import { Injectable, inject } from '@angular/core';
import { HttpClient } from '@angular/common/http';
import { firstValueFrom } from 'rxjs';
import { Order, PaymentStatus } from '../checkout/checkout.models';

@Injectable({ providedIn: 'root' })
export class OrderService {
  private readonly http = inject(HttpClient);

  async getOrderById(orderId: string): Promise<Order> {
    return firstValueFrom(
      this.http.get<Order>(`/api/orders/${orderId}`)
    );
  }

  async getOrdersByBuyer(buyerId: string): Promise<Order[]> {
    return firstValueFrom(
      this.http.get<Order[]>(`/api/orders/buyer/${buyerId}`)
    );
  }

  async getPaymentStatus(orderId: string): Promise<PaymentStatus> {
    return firstValueFrom(
      this.http.get<PaymentStatus>(`/api/payments/order/${orderId}`)
    );
  }
}
```
