# 7.4.5 — SignalR Real-Time Order Updates

> **Ref**: [plans/11-frontend-angular.md](../../plans/11-frontend-angular.md) · [plans/07-realtime-signalr.md](../../plans/07-realtime-signalr.md)

## Goal
Integrate SignalR to receive real-time order status updates. When the Ordering saga transitions states, the Notification.Worker pushes updates to the frontend via SignalR. The frontend updates the `OrderStore` and `CheckoutStore` automatically.

## Prerequisites
- Sub-plan 7.4.2 completed (OrderStore, CheckoutStore)
- Phase 5 backend (Notification.Worker with SignalR hub) will be implemented later — for now we stub the connection
- `@microsoft/signalr` package installed (from 7.0.1)

## Steps

### Step 1: Create Order Update model

File: `src/web/src/app/core/signalr/signalr.models.ts`
```typescript
export interface OrderUpdate {
  orderId: string;
  correlationId: string;
  status: string;
  timestamp: string;
  reason?: string;
}
```

### Step 2: Create Notification Service

File: `src/web/src/app/core/signalr/notification.service.ts`
```typescript
import { Injectable, inject, signal } from '@angular/core';
import { HubConnection, HubConnectionBuilder, LogLevel } from '@microsoft/signalr';
import { OrderUpdate } from './signalr.models';

@Injectable({ providedIn: 'root' })
export class NotificationService {
  private hubConnection: HubConnection | null = null;

  readonly connected = signal(false);
  readonly lastUpdate = signal<OrderUpdate | null>(null);

  async start(): Promise<void> {
    if (this.hubConnection) return;

    this.hubConnection = new HubConnectionBuilder()
      .withUrl('/hubs/notifications', { withCredentials: true })
      .withAutomaticReconnect()
      .configureLogging(LogLevel.Information)
      .build();

    this.hubConnection.on('OrderUpdate', (update: OrderUpdate) => {
      this.lastUpdate.set(update);
    });

    this.hubConnection.onreconnecting(() => this.connected.set(false));
    this.hubConnection.onreconnected(() => this.connected.set(true));

    try {
      await this.hubConnection.start();
      this.connected.set(true);
    } catch (err) {
      console.error('SignalR connection failed:', err);
      this.connected.set(false);
    }
  }

  async stop(): Promise<void> {
    if (this.hubConnection) {
      await this.hubConnection.stop();
      this.hubConnection = null;
      this.connected.set(false);
    }
  }
}
```

import { initSignalR } from './core/signalr/signalr.initializer';

### Step 5: Connect SignalR updates to OrderStore and CheckoutStore

Create a reactive bridge that listens to `NotificationService.lastUpdate` and updates stores:

File: `src/web/src/app/core/signalr/order-update.handler.ts`
```typescript
import { Injectable, inject, effect } from '@angular/core';
import { NotificationService } from './notification.service';
import { OrderStore } from '../../features/orders/order.store';
import { CheckoutStore } from '../../features/checkout/checkout.store';
import { OrderStatus } from '../../features/checkout/checkout.models';

@Injectable({ providedIn: 'root' })
export class OrderUpdateHandler {
  private readonly notificationService = inject(NotificationService);
  private readonly orderStore = inject(OrderStore);
  private readonly checkoutStore = inject(CheckoutStore);

  constructor() {
    effect(() => {
      const update = this.notificationService.lastUpdate();
      if (!update) return;

      const status = update.status as OrderStatus;

      // Update order store (order history list)
      this.orderStore.updateOrderStatus(update.orderId, status);

      // Update checkout store if this is the current checkout
      if (this.checkoutStore.correlationId() === update.correlationId) {
        // Reload the full order to get updated data
        this.orderStore.loadOrderById(update.orderId);
      }
    });
  }
}
```

### Step 6: Initialize OrderUpdateHandler in `app.config.ts`

Add to the providers in `src/web/src/app/app.config.ts`:

```typescript
import { OrderUpdateHandler } from './core/signalr/order-update.handler';

// In provideAppInitializer:
provideAppInitializer(() => {
  if (isPlatformBrowser(inject(PLATFORM_ID))) {
    void inject(AuthStore).checkAuth();
    void inject(NotificationService).start();
    // Initialize the handler to bridge SignalR updates to stores
    inject(OrderUpdateHandler);
  }
}),
```
