# 3.5 — Cart.API

> **Ref**: [plans/02-domain-decomposition.md](../../plans/02-domain-decomposition.md)

## Goal
Create the Cart service as a thin API directly interacting with Redis. No Domain/Application layers needed. Endpoints for getting the cart, updating items, and checkout (publishing `OrderSubmittedEvent` via MassTransit).

## Prerequisites
- Sub-plan 3.0 completed (Cart contracts)

namespace Cart.API.Repositories;

public interface ICartRepository
{
    Task<ShoppingCart> GetCartAsync(string buyerId, CancellationToken ct = default);
    Task<ShoppingCart> UpdateCartAsync(ShoppingCart cart, CancellationToken ct = default);
    Task DeleteCartAsync(string buyerId, CancellationToken ct = default);

File: `src/Microservices/Cart/Cart.API/Repositories/CartRepository.cs`
```csharp
using System.Text.Json;
using Cart.API.Models;
using Microsoft.Extensions.Caching.Distributed;

namespace Cart.API.Repositories;

public class CartRepository(IDistributedCache cache) : ICartRepository
{
    public async Task<ShoppingCart> GetCartAsync(string buyerId, CancellationToken ct = default)
    {
        var data = await cache.GetStringAsync(buyerId, ct);
        if (string.IsNullOrEmpty(data))
        {
            return new ShoppingCart(buyerId);
        }

        return JsonSerializer.Deserialize<ShoppingCart>(data) ?? new ShoppingCart(buyerId);
    }

    public async Task<ShoppingCart> UpdateCartAsync(ShoppingCart cart, CancellationToken ct = default)
    {
        var options = new DistributedCacheEntryOptions
        {
            SlidingExpiration = TimeSpan.FromDays(7)
        };

        var data = JsonSerializer.Serialize(cart);
        await cache.SetStringAsync(cart.BuyerId, data, options, ct);

        return cart;
    }

    public async Task DeleteCartAsync(string buyerId, CancellationToken ct = default)
    {
        await cache.RemoveAsync(buyerId, ct);
    }
}
```

### Step 4: Create Cart Endpoints

File: `src/Microservices/Cart/Cart.API/Endpoints/CartEndpoints.cs`
```csharp
using BuildingBlocks.SharedContracts.Dtos;
using BuildingBlocks.SharedContracts.Events.Cart;
using Cart.API.Models;
using Cart.API.Repositories;
using MassTransit;
using Microsoft.AspNetCore.Mvc;

namespace Cart.API.Endpoints;

public static class CartEndpoints
{
    public static void MapCartEndpoints(this IEndpointRouteBuilder app)
    {
        var group = app.MapGroup("/api/cart")
            .WithTags("Cart")
            .WithOpenApi();

        // For demo, we get BuyerId from a header. In reality, it comes from claims.
        group.MapGet("/", async (
            [FromHeader(Name = "x-buyer-id")] string buyerId,
            [FromServices] ICartRepository repository,
            CancellationToken ct) =>
        {
            if (string.IsNullOrEmpty(buyerId)) return Results.BadRequest("Buyer ID required.");
            var cart = await repository.GetCartAsync(buyerId, ct);
            return Results.Ok(cart);
        });

        group.MapPost("/", async (
            [FromHeader(Name = "x-buyer-id")] string buyerId,
            [FromBody] ShoppingCart cartUpdate,
            [FromServices] ICartRepository repository,
            CancellationToken ct) =>
        {
            if (string.IsNullOrEmpty(buyerId)) return Results.BadRequest("Buyer ID required.");
            cartUpdate.BuyerId = buyerId;
            var updated = await repository.UpdateCartAsync(cartUpdate, ct);
            return Results.Ok(updated);
        });

        group.MapDelete("/", async (
            [FromHeader(Name = "x-buyer-id")] string buyerId,
            [FromServices] ICartRepository repository,
            CancellationToken ct) =>
        {
            if (string.IsNullOrEmpty(buyerId)) return Results.BadRequest("Buyer ID required.");
            await repository.DeleteCartAsync(buyerId, ct);
            return Results.NoContent();
        });

        group.MapPost("/checkout", async (
            [FromHeader(Name = "x-buyer-id")] string buyerId,
            [FromServices] ICartRepository repository,
            [FromServices] IPublishEndpoint publishEndpoint,
            CancellationToken ct) =>
        {
            if (string.IsNullOrEmpty(buyerId)) return Results.BadRequest("Buyer ID required.");
            var cart = await repository.GetCartAsync(buyerId, ct);

            if (cart.Items.Count == 0) return Results.BadRequest("Cart is empty.");

            var correlationId = Guid.NewGuid();
            var itemsContract = cart.Items.Select(i => new OrderItemContract(i.Sku, i.Quantity)).ToList();

            var orderSubmittedEvent = new OrderSubmittedEvent(
                correlationId,
                buyerId,
                itemsContract,
                DateTime.UtcNow
            );

            await publishEndpoint.Publish(orderSubmittedEvent, ct);

            // Optionally clear cart or leave it to be cleared after order succeeds
            await repository.DeleteCartAsync(buyerId, ct);

            return Results.Accepted(value: new { CorrelationId = correlationId });
        });
    }
}
```

### Step 5: Configure `Program.cs`

File: `src/Microservices/Cart/Cart.API/Program.cs`
```csharp
using Cart.API.Endpoints;
using Cart.API.Repositories;
using MassTransit;

var builder = WebApplication.CreateBuilder(args);

builder.AddServiceDefaults();

// Redis
builder.AddRedisDistributedCache("redis");

builder.Services.AddScoped<ICartRepository, CartRepository>();

// MassTransit
builder.Services.AddMassTransit(x =>
{
    x.UsingRabbitMq((context, cfg) =>
    {
        var configuration = context.GetRequiredService<IConfiguration>();
        var connectionString = configuration.GetConnectionString("messaging");
        
        if (!string.IsNullOrEmpty(connectionString))
        {
            cfg.Host(connectionString);
        }

        cfg.ConfigureEndpoints(context);
    });
});

builder.Services.AddEndpointsApiExplorer();
builder.Services.AddSwaggerGen();

var app = builder.Build();

app.MapDefaultEndpoints();

if (app.Environment.IsDevelopment())
{
    app.UseSwagger();
    app.UseSwaggerUI();
}

app.MapCartEndpoints();

app.Run();
```
